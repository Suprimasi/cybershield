            function generateCryptoPassword() {
                const length = parseInt(lengthSlider.value, 10);
                let charset = '';
                const upperChars = 'ABCDEFGHJKLMNPQRSTUVWXYZ';
                const lowerChars = 'abcdefghijkmnopqrstuvwxyz';
                const numberChars = '23456789';
                const symbolChars = '!@#$%^&*()_+-=[]{}|;:,.?';

                const mandatoryChars = [];
                const randomBuffer = new Uint32Array(length + 4);
                window.crypto.getRandomValues(randomBuffer);
                let randIdx = 0;

                if (genUpper.checked) {
                    charset += upperChars;
                    mandatoryChars.push(upperChars[randomBuffer[randIdx++] % upperChars.length]);
                }
                if (genLower.checked) {
                    charset += lowerChars;
                    mandatoryChars.push(lowerChars[randomBuffer[randIdx++] % lowerChars.length]);
                }
                if (genNumbers.checked) {
                    charset += numberChars;
                    mandatoryChars.push(numberChars[randomBuffer[randIdx++] % numberChars.length]);
                }
                if (genSymbols.checked) {
                    charset += symbolChars;
                    mandatoryChars.push(symbolChars[randomBuffer[randIdx++] % symbolChars.length]);
                }

                if (!charset) {
                    genLower.checked = true;
                    charset = lowerChars;
                    showToast('Нужно оставить хотя бы один тип символов');
                }

                const resultChars = [...mandatoryChars];
                while (resultChars.length < length) {
                    resultChars.push(charset[randomBuffer[randIdx++] % charset.length]);
                }

                const shuffleBuffer = new Uint32Array(resultChars.length);
                window.crypto.getRandomValues(shuffleBuffer);
                for (let i = resultChars.length - 1; i > 0; i--) {
                    const j = shuffleBuffer[i] % (i + 1);
                    [resultChars[i], resultChars[j]] = [resultChars[j], resultChars[i]];
                }

                const finalPwd = resultChars.join('');
                passwordInput.value = finalPwd;
                analyzePassword(finalPwd);
            }

            const kbTabs = document.querySelectorAll('.kb-tab');
            const kbContent = document.getElementById('kb-content');

            function renderKnowledgeTab(topicKey) {

                void kbContent.offsetWidth;

                kbContent.innerHTML = `

                        <div class="lg:col-span-5 bg-cyber-bg/80 border border-cyber-border rounded-xl p-5 sm:p-6">

                                    <div class="w-7 h-7 rounded-lg bg-cyber-cyan/10 border border-cyber-cyan/30 text-cyber-cyan font-mono font-bold text-xs flex items-center justify-center shrink-0 mt-0.5">

            function updateHygieneMetrics() {
                const total = hygieneSteps.length;
                const completed = hygieneSteps.filter(s => checkedState[s.id]).length;
                const percent = Math.round((completed / total) * 100);

                hygieneProgressBar.style.width = `${percent}%`;
                hygienePercentEl.textContent = `${percent}%`;
                headerScoreEl.textContent = `${percent}%`;
                hygieneCountEl.textContent = `${completed} из ${total} сделано`;

                if (percent === 0) {
                    hygieneStatusTitle.textContent = 'Начальный уровень (много рисков)';
                    hygieneStatusTitle.className = 'text-xl sm:text-2xl font-bold text-cyber-danger';
                    hygieneAdvice.textContent = 'Начни с самого простого: поставь двухфакторную защиту на почту и облачный пароль в Телеграме.';
                } else if (percent < 50) {
                    hygieneStatusTitle.textContent = 'Базовый уровень (можно лучше)';
                    hygieneStatusTitle.className = 'text-xl sm:text-2xl font-bold text-amber-400';
                    hygieneAdvice.textContent = 'Уже неплохо, но стоит разделить пароли для разных сайтов и начать пользоваться менеджером паролей.';
                } else if (percent < 100) {
                    hygieneStatusTitle.textContent = 'Хороший уровень (уверенная защита)';
                    hygieneStatusTitle.className = 'text-xl sm:text-2xl font-bold text-cyber-cyan';
                    hygieneAdvice.textContent = 'Ты защищён лучше большинства пользователей. Осталось добить пару пунктов до полного спокойствия.';
                } else {
                    hygieneStatusTitle.textContent = 'Максимальный уровень (всё настроено)';
                    hygieneStatusTitle.className = 'text-xl sm:text-2xl font-bold text-cyber-emerald';
                    hygieneAdvice.textContent = 'Отлично! Все базовые правила безопасности соблюдены.';
                }

                try {
                    localStorage.setItem('cybershield_checklist', JSON.stringify(checkedState));
                } catch (e) {}
            }

            function renderChecklist() {
                checklistContainer.innerHTML = hygieneSteps.map(step => {
                    const isChecked = !!checkedState[step.id];
                    return `
                        <label for="${step.id}" class="p-4 rounded-2xl border transition cursor-pointer flex items-start gap-3.5 select-none ${isChecked ? 'bg-cyber-emerald/10 border-cyber-emerald/50' : 'glass-panel border-cyber-border hover:border-slate-500'}">
                            <input type="checkbox" id="${step.id}" ${isChecked ? 'checked' : ''} class="mt-1 w-5 h-5 accent-cyber-emerald rounded cursor-pointer shrink-0">
                            <div>
                                <div class="font-semibold text-sm sm:text-base ${isChecked ? 'text-white' : 'text-slate-200'}">${step.title}</div>
                                <div class="text-xs text-slate-400 mt-1 leading-relaxed">${step.desc}</div>
                            </div>
                        </label>
                    `;
                }).join('');

                hygieneSteps.forEach(step => {
                    const checkbox = document.getElementById(step.id);
                    checkbox.addEventListener('change', (e) => {
                        checkedState[step.id] = e.target.checked;
                        renderChecklist();
                        updateHygieneMetrics();
                    });
                });
            }

            document.getElementById('reset-checklist-btn').addEventListener('click', () => {
                checkedState = {};
                renderChecklist();
                updateHygieneMetrics();
                showToast('Отметки сброшены');
            });

            renderChecklist();
            updateHygieneMetrics();
        });
    </script>
